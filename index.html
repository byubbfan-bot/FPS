import java.util.Random; import java.util.Scanner; public class Main { static class Tier { String name; int threshold; int pitcherDifficulty; Tier(String name, int threshold, int pitcherDifficulty) { this.name = name; this.threshold = threshold; this.pitcherDifficulty = pitcherDifficulty; } } static final Tier[] TIERS = { new Tier("Single-A", 120, 45), new Tier("Double-A", 260, 60), new Tier("Triple-A", 420, 75), new Tier("MLB Majors", 650, 88), new Tier("MLB All-Star", 9999, 95) }; static class Player { String name; String position; int contact = 45; int power = 40; int discipline = 40; int speed = 50; int energy = 100; int hype = 10; int careerHits = 0; int careerHrs = 0; int careerRbi = 0; int atBats = 0; int overallPoints = 0; int tierIdx = 0; Player(String name, String position) { this.name = name; this.position = position; } String getCurrentTier() { return TIERS[this.tierIdx].name; } double getBattingAvg() { return (this.atBats > 0) ? ((double) this.careerHits / this.atBats) : 0.0; } boolean checkPromotion() { if (this.tierIdx < TIERS.length - 1) { int req = TIERS[this.tierIdx].threshold; if (this.overallPoints >= req) { this.tierIdx++; System.out.println("\n========================================================"); System.out.println("🚨 CALL-UP ALERT! You have been promoted to " + getCurrentTier() + "!"); System.out.println("========================================================\n"); return true; } } return false; } } private static final Random rng = new Random(); private static final Scanner scanner = new Scanner(System.in); public static void main(String[] args) { System.out.println("============================================================"); System.out.println(" ROAD TO COOPERSTOWN: AN MLB JOURNEY "); System.out.println("============================================================"); System.out.print("Enter your prospect's name (Default: Kid Carter): "); String name = scanner.nextLine().trim(); if (name.isEmpty()) name = "Kid Carter"; System.out.print("Enter position (e.g., SS, OF, 3B - Default: SS): "); String pos = scanner.nextLine().trim().toUpperCase(); if (pos.isEmpty()) pos = "SS"; Player player = new Player(name, pos); System.out.println("\nDrafted in the MLB Draft! Assigned to Single-A ball."); boolean running = true; while (running) { if (player.tierIdx >= TIERS.length - 1 && player.overallPoints >= 900) { System.out.println("\n************************************************************"); System.out.println("CONGRATULATIONS! " + player.name + " is an MLB All-Star and Legend!"); System.out.println("************************************************************"); showStatus(player); break; } showStatus(player); System.out.println("\nACTIONS:"); System.out.println("1. Play Game Day (-20 Energy)"); System.out.println("2. Facility Training (-25 Energy)"); System.out.println("3. Rest & Recovery"); System.out.println("4. Retire / Exit"); System.out.print("\nChoose action > "); String choice = scanner.nextLine().trim(); switch (choice) { case "1": playGame(player); break; case "2": train(player); break; case "3": rest(player); break; case "4": System.out.println("\n" + player.name + " hung up the spikes. Thanks for playing!"); running = false; break; default: System.out.println("Invalid input. Please choose 1, 2, 3, or 4."); } } } private static void showStatus(Player p) { int avgInt = (int) Math.round(p.getBattingAvg() * 1000); String avgStr = String.format(".%03d", avgInt); System.out.println("\n" + "=".repeat(50)); System.out.println("PLAYER: " + p.name + " | POS: " + p.position + " | LEVEL: " + p.getCurrentTier()); System.out.println("ENERGY: " + p.energy + "/100 | HYPE: " + p.hype + " | PROGRESS PTS: " + p.overallPoints); System.out.println("STATS: Contact: " + p.contact + " | Power: " + p.power + " | Eye: " + p.discipline + " | Speed: " + p.speed); System.out.println("RECORD: " + p.careerHits + " H / " + p.atBats + " AB (" + avgStr + ") | HR: " + p.careerHrs + " | RBI: " + p.careerRbi); System.out.println("=".repeat(50)); } private static void train(Player p) { if (p.energy < 25) { System.out.println("\nYou're exhausted. Rest first to avoid injury!"); return; } System.out.println("\nSelect Training Focus:"); System.out.println("1. Cage Work (+Contact, +XP)"); System.out.println("2. Weight Room (+Power, -Speed risk)"); System.out.println("3. Video Study (+Discipline, +XP)"); System.out.println("4. Sprint Drills (+Speed)"); System.out.print("> "); String choice = scanner.nextLine().trim(); p.energy -= 25; switch (choice) { case "1": { int gain = rng.nextInt(3) + 2; // 2-4 p.contact += gain; p.overallPoints += 15; System.out.println("\nLocked in mechanics. Contact increased by +" + gain + "!"); break; } case "2": { int gain = rng.nextInt(4) + 2; // 2-5 p.power += gain; if (rng.nextDouble() < 0.25 && p.speed > 35) { p.speed -= 1; System.out.println("\nHeavy sets finished. Power +" + gain + ", but lost 1 Speed to muscle mass."); } else { System.out.println("\nHeavy sets finished. Power +" + gain + "!"); } p.overallPoints += 15; break; } case "3": { int gain = rng.nextInt(3) + 3; // 3-5 p.discipline += gain; p.overallPoints += 20; System.out.println("\nStudied release angles and pitch tunnels. Discipline +" + gain + "!"); break; } case "4": { int gain = rng.nextInt(3) + 2; // 2-4 p.speed += gain; p.overallPoints += 10; System.out.println("\nExplosive base-running drills. Speed +" + gain + "!"); break; } default: System.out.println("\nIndecision wasted the session. Energy spent without gains."); } } private static void rest(Player p) { int recovered = rng.nextInt(16) + 35; // 35-50 p.energy = Math.min(100, p.energy + recovered); System.out.println("\nIce bath and full night of sleep. Energy restored to " + p.energy + "%."); } private static void simulateAtBat(Player p) { int diff = TIERS[p.tierIdx].pitcherDifficulty; int balls = 0; int strikes = 0; System.out.println("\n--- Stepping into the box vs. " + p.getCurrentTier() + " Pitching (OVR " + diff + ") ---"); while (strikes < 3 && balls < 4) { boolean pitchInZone = rng.nextDouble() < 0.60; int pitchQuality = diff + (rng.nextInt(25) - 12); System.out.println("\nCount: " + balls + "-" + strikes); System.out.println("1. Contact Swing | 2. Power Swing | 3. Take Pitch"); System.out.print("Approach > "); String approach = scanner.nextLine().trim(); if (approach.equals("3")) { if (!pitchInZone) { if ((p.discipline + (rng.nextInt(25) + 1)) > (diff * 0.72)) { System.out.println(">> Good take outside. Ball."); balls++; } else { System.out.println(">> Chased out of the zone! Strike."); strikes++; } } else { System.out.println(">> Called strike on the corner."); strikes++; } } else if (approach.equals("1") || approach.equals("2")) { boolean isPower = approach.equals("2"); int statMod = isPower ? p.power : p.contact; int penalty = isPower ? 18 : 5; int roll = statMod + (rng.nextInt(40) + 1) - penalty; if (!pitchInZone) roll -= 22; if (roll < pitchQuality) { System.out.println(">> Swing and a miss!"); strikes++; } else { p.atBats++; int outcomeRoll = roll - pitchQuality + (rng.nextInt(30) + 1); if (isPower && outcomeRoll > 32) { System.out.println("\n🔥 CRUSHED TO THE UPPER DECK! HOME RUN!"); p.careerHits++; p.careerHrs++; p.careerRbi += rng.nextInt(3) + 1; p.overallPoints += 45; p.hype += 15; } else if (outcomeRoll > 18) { System.out.println("\n⚾ Driven into the gap for extra bases!"); p.careerHits++; p.careerRbi += rng.nextInt(3); p.overallPoints += 25; p.hype += 8; } else if (outcomeRoll >= 4) { System.out.println("\n⚾ Sharp ground ball through the infield for a single."); p.careerHits++; p.careerRbi += rng.nextInt(2); p.overallPoints += 15; p.hype += 4; } else { System.out.println("\nRoutine grounder to shortstop. Thrown out at first."); } return; } } else { System.out.println("Frozen looking! Strike called."); strikes++; } } p.atBats++; if (balls == 4) { System.out.println("\n>> Ball four! Walk drawn."); p.overallPoints += 10; p.hype += 2; } else { System.out.println("\n>> Strike three. Down on strikes."); p.hype = Math.max(0, p.hype - 3); } } private static void playGame(Player p) { if (p.energy < 20) { System.out.println("\nManager benched you today due to dead legs. Rest up."); return; } p.energy -= 20; System.out.println("\nGame Day under the lights in " + p.getCurrentTier() + "!"); for (int ab = 1; ab <= 3; ab++) { System.out.println("\n[ Plate Appearance " + ab + "/3 ]"); simulateAtBat(p); try { Thread.sleep(300); } catch (InterruptedException ignored) {} } p.checkPromotion(); } }
